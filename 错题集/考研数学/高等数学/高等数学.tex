\chapter{高等数学}

\section{题目}
\begin{enumerate}
    \item \bt 已知$y=C_1+C_2 \sin{x} + e^x$为某二阶微分方程解,则原二阶微分方程式\_\_\_\_\_ 

    \item \bl 设函数f(x)可导,且对任意常数x,h满足$f(x+h)=\int_{x}^{x+h}t\left[f(t+h)+t^2\right]\d t + f(x)$, 
    且$\lim_{x\to 0}\left[1+f(x)\right]^{\frac{1}{x^2}}=a(a>0)$求$f(x)$的表达式及常数a. 
\end{enumerate}


\section{解答}

\begin{enumerate}
    \item 这道题比较特殊,不是通过解的结构去反推二阶微分方程,首先题目没有说{\color{red} 这是一个二阶常系数微分方程}
    而是只说它是一个微分方程. 那就考虑直接对等式两边求导, 消去参数. 
    \begin{align*}
        y' &= C_2 \cos{x} + e^x \\
        y'' &= -C_2 \sin{x} + e^x 
    \end{align*}
    解出$C_2$带入$y''$表达式即可求出最终二阶微分方程式 
    $$
    y'' + \tan{x} y' = e^x(1+\tan{x})
    $$ 

    \item 这道题就比较难了, 主要难在如何构造出f(x)的微分方程. 
    \begin{solution}[方法一]
    \begin{align*}
        f(x+h) &= \int_{x}^{x+h}t\left[f(t+h)+t^2\right]\d t \\
        &= \int_{x-h}^{x}tf(t)\d t - h\int_{x-h}^{x}f(t)\d t + \int_{x}^{x+h}t^3\d t 
    \end{align*}
    {\color{red} 左右两边同时对h求导,且令h=0}可得到关于x的微分方程 
    $$
    f'(x) = xf(x) + x^3
    $$
    \end{solution}

    \begin{solution}[方法二: 导数定义+积分中值定理]
    \begin{align*}
        f(x+h) - f(x) &= \int_{x}^{x+h}t\left[f(t+h)+t^2\right]\d t \\
        \lim_{h\to 0}\frac{f(x+h) - f(x)}{h} &= \frac{1}{h} \int_{x}^{x+h}t[f(t+h)+t^2]\d t \\
        &= \lim_{h\to 0}\frac{1}{h} (x+ h - x) \delta(f(\delta + h) + \delta^2), (x < \delta < x + h) \\
        &= xf(x) + x^3
    \end{align*}
    即$f'(x) = xf(x)+x^3$
    \end{solution}


    \item 在一棵含有n个关键字的m阶B树中进行查找, 至多读盘(\qquad)次.
\end{enumerate}
